\begin{frame}{Metadata và audit contract}
\begin{itemize}
  \item \texttt{ctl.ETLBatch}: một batch chung, trạng thái và tổng row counts.
  \item \texttt{ctl.ETLFileLoad}: từng file, \texttt{FileLoadID}, trạng thái, accepted/rejected.
  \item \texttt{ctl.DataQualityResult}: test code, expected, actual và PASS/FAIL/WARN.
  \item Điều kiện bất biến: \texttt{RowsRead = RowsAccepted + RowsRejected}.
  \item Không đóng batch \texttt{SUCCEEDED} khi file audit hoặc DQ còn lỗi.
\end{itemize}
\end{frame}

\begin{frame}{Dashboard specification}
\begin{enumerate}
  \item Executive overview: application, default và phân khúc chính.
  \item Credit risk profile: default theo age, income, affordability, contract.
  \item Payment behavior: late rate, days late và scheduled/paid amount.
  \item Credit history: previous decision, active bureau, debt và overdue.
  \item Data quality: batch status, source counts, rejects, sentinels và orphans.
\end{enumerate}
